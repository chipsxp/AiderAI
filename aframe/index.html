AFRAME.registerComponent('cursor-listener', { init: function () { var el = this.el; el.addEventListener('click', function () { console.log('Model clicked!'); }); el.addEventListener('mouseenter', function () { console.log('Mouse entered model'); el.pause(); // This pauses all animations on the entity }); el.addEventListener('mouseleave', function () { console.log('Mouse left model'); el.play(); // This resumes all animations on the entity }); } }); A-Frame VR Experience
